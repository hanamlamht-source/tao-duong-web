        // Real-time Clock
        function updateClock() {
            const now = new Date();
            const days = ['Chủ Nhật', 'Thứ Hai', 'Thứ Ba', 'Thứ Tư', 'Thứ Năm', 'Thứ Sáu', 'Thứ Bảy'];
            const dayName = days[now.getDay()];
            const dateStr = `${dayName}, ${now.getDate()}/${now.getMonth() + 1}/${now.getFullYear()} - ${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}:${String(now.getSeconds()).padStart(2, '0')}`;
            const clockEl = document.getElementById('realtime-clock');
            if(clockEl) clockEl.innerHTML = `<i class="far fa-clock mr-1"></i> ${dateStr}`;
        }

        // Carousel Logic
        function showSlide(index) {
            const slides = document.querySelectorAll('.carousel-slide');
            if(slides.length === 0) return;
            slides.forEach((slide, idx) => {
                slide.style.opacity = idx === index ? '1' : '0';
            });
        }
        function nextSlide() {
            const slides = document.querySelectorAll('.carousel-slide');
            currentSlide = (currentSlide + 1) % slides.length;
            showSlide(currentSlide);
        }
        function prevSlide() {
            const slides = document.querySelectorAll('.carousel-slide');
            currentSlide = (currentSlide - 1 + slides.length) % slides.length;
            showSlide(currentSlide);
        }

        // Modal Controls
        function openModal(title, bodyHtml) {
            document.getElementById('modal-title').innerText = title;
            document.getElementById('modal-body').innerHTML = bodyHtml;
            document.getElementById('modal-overlay').classList.remove('hidden');
        }

        function openNewsModal(id) {

                    </div>
                `;
                openModal(news.title, html);

        // Search Filter
        function handleSearch() {
            const query = document.getElementById('search-input').value.toLowerCase().trim();
            if(!query) {
                renderNews(newsData);
                return;
            }
            const filtered = newsData.filter(n => 
                n.title.toLowerCase().includes(query) || 
                n.summary.toLowerCase().includes(query) ||
                n.category.toLowerCase().includes(query)
            );
            renderNews(filtered);
            scrollToSection('tin-tuc');
        }

        // Accessibility: Font Resizer
        function changeFontSize(delta) {
            fontSizeLevel = Math.max(-1, Math.min(2, fontSizeLevel + delta));
            const body = document.getElementById('body-content');
            if(fontSizeLevel === -1) body.style.fontSize = '14px';
            else if(fontSizeLevel === 0) body.style.fontSize = '16px';
            else if(fontSizeLevel === 1) body.style.fontSize = '18px';
            else if(fontSizeLevel === 2) body.style.fontSize = '20px';
        }

        function resetFontSize() {
            fontSizeLevel = 0;
            document.getElementById('body-content').style.fontSize = '16px';
        }

        // Dark Mode Toggle
        function toggleDarkMode() {
            const html = document.documentElement;
            const icon = document.getElementById('theme-icon');
            if(html.classList.contains('dark')) {
                html.classList.remove('dark');
                icon.className = 'fas fa-moon';
                localStorage.setItem('theme', 'light');
            } else {
                html.classList.add('dark');
                icon.className = 'fas fa-sun';
                localStorage.setItem('theme', 'dark');
            }
        }

        // Feedback System with LocalStorage
        function submitFeedback(e) {
            e.preventDefault();
            const name = document.getElementById('fb-name').value.trim();
            const phone = document.getElementById('fb-phone').value.trim();
            const topic = document.getElementById('fb-topic').value;
            const content = document.getElementById('fb-content').value.trim();

            if(!name || !phone || !content) return;

            const feedbackItem = {
                id: Date.now(),
                name,
                phone,
                topic,
                content,
                date: new Date().toLocaleDateString('vi-VN')
            };

            const existing = JSON.parse(localStorage.getItem('tao_duong_feedbacks') || '[]');
            existing.unshift(feedbackItem);
            localStorage.setItem('tao_duong_feedbacks', JSON.stringify(existing));

            openModal('Gửi Phản Ánh Thành Công', `

            `);

            document.getElementById('feedback-form').reset();
            renderPublicFeedback();
        }

        function renderPublicFeedback() {
            const listEl = document.getElementById('public-feedback-list');
            if(!listEl) return;

            const items = JSON.parse(localStorage.getItem('tao_duong_feedbacks') || '[]');
            if(items.length === 0) {
                listEl.innerHTML = `<p class="text-gray-400 italic text-[11px]">Chưa có phản ánh nào gần đây.</p>`;
                return;
            }

            listEl.innerHTML = items.slice(0, 5).map(item => `
                <div class="p-2 bg-white dark:bg-gray-700 rounded border border-gray-200 dark:border-gray-600">
                    <div class="flex justify-between items-center text-[10px] text-gray-400">

                        <span>${item.date}</span>

                    </span>
                    <p class="text-[11px] text-gray-600 dark:text-gray-300 mt-1 line-clamp-2">${item.content}</p>
                </div>
            `).join('');
        }
